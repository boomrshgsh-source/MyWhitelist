-- Highlight
local highlight
function enabled_highlight()
    if highlight then highlight:Destroy() end
    highlight = Instance.new("Highlight")
    highlight.Parent = LocalPlayer.Character
    highlight.FillColor = Color3.fromRGB(0, 255, 253)
    highlight.FillTransparency = 0.5
    highlight.OutlineColor = Color3.fromRGB(0, 0, 0)
    highlight.OutlineTransparency = 0
end

function disable_highlight()
    if highlight then
        highlight:Destroy()
        highlight = nil
    end
end

function enabled_noclip()
    local character = LocalPlayer.Character
    if not character then return end
    for _, part in pairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            if part.Name == "LowerTorso" or part.Name == "UpperTorso" or part.Name == "HumanoidRootPart" then
                part.CanCollide = false
            end
        end
    end
end

function disable_noclip()
    local character = LocalPlayer.Character
    if not character then return end
    for _, part in pairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            if part.Name == "LowerTorso" or part.Name == "UpperTorso" or part.Name == "HumanoidRootPart" then
                part.CanCollide = true
            end
        end
    end
end

-- Create BodyVelocity for flying
function createFlightObjects()
    if bodyVelocity then bodyVelocity:Destroy() end

    bodyVelocity = Instance.new("BodyVelocity")
    bodyVelocity.Name = "You're so crazy, manipulate me Fucking chase me, fucking break_ me You're my everything, please just rape me"
    bodyVelocity.MaxForce = Vector3.new(9e9, 9e9, 9e9)
    bodyVelocity.Velocity = Vector3.zero
    bodyVelocity.Parent = hrp
end

function destroyFlightObjects()
    if bodyVelocity then
        bodyVelocity:Destroy()
        bodyVelocity = nil
    end
end

-- Main tween loop
function startTweenLoop()
    if tweenConnection then return end
    
    createFlightObjects()
    
    tweenConnection = RunService.Heartbeat:Connect(function(dt)
        if not isTweenEnabled or not hrp or not hrp.Parent then
            if bodyVelocity then bodyVelocity.Velocity = Vector3.zero end
            return
        end

        if not currentTarget then
            if bodyVelocity then bodyVelocity.Velocity = Vector3.zero end
            return
        end

        local direction = currentTarget - hrp.Position
        local distanceToTarget = direction.Magnitude

        -- Warp if close
        if distanceToTarget < 130 then
            hrp.CFrame = CFrame.new(currentTarget)
            if bodyVelocity then bodyVelocity.Velocity = Vector3.zero end
            return
        end

        -- Move with velocity
        if bodyVelocity then
            local moveDist = math.min(speed * dt, distanceToTarget)
            local newPos = hrp.Position + direction.Unit * moveDist
            hrp.CFrame = CFrame.lookAt(newPos, currentTarget)
            bodyVelocity.Velocity = direction.Unit * speed
        end
    end)
end

function stopTweenLoop()
    isTweenEnabled = false
    currentTarget = nil
    FollowMob = nil
    destroyFlightObjects()
    disable_highlight()
    if tweenConnection then
        tweenConnection:Disconnect()
        tweenConnection = nil
    end
end

function equip_weapon(type_of_weapon)
    local char = LocalPlayer.Character
    local backpack = LocalPlayer.Backpack
    if not char then return end
    local tool_to_equip
    local held_tool = char:FindFirstChildOfClass("Tool")
    if held_tool and held_tool.ToolTip == type_of_weapon then
        tool_to_equip = held_tool
    else
        for _, tool in ipairs(backpack:GetChildren()) do
            if tool:IsA("Tool") and tool.ToolTip == type_of_weapon then
                tool_to_equip = tool
                char.Humanoid:EquipTool(tool)
                break
            end
        end
    end
    current_weapon = tool_to_equip
end

local FollowMobs = {}

function BringMob(target)
    if target and target:FindFirstChild("HumanoidRootPart") then
        local hrp = target.HumanoidRootPart
        if not hrp:FindFirstChild("BringBV") then
            local bv = Instance.new("BodyVelocity")
            bv.Name = "BringBV"
            bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
            bv.Velocity = Vector3.zero
            bv.Parent = hrp
        end
        table.insert(FollowMobs, target)
    end
end

game:GetService("RunService").Heartbeat:Connect(function()
    local playerHrp = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not playerHrp then return end

    for i = #FollowMobs, 1, -1 do
        local mob = FollowMobs[i]
        if mob and mob.Parent and mob:FindFirstChild("HumanoidRootPart") then
            local hrp = mob.HumanoidRootPart
            local bv = hrp:FindFirstChild("BringBV")
            if bv then
                local targetPos = playerHrp.Position + Vector3.new(0, -distance, 0)
                local direction = targetPos - hrp.Position
                local distance = direction.Magnitude

                if distance > 1 then
                    local speed = math.min(distance * 2, 300)
                    bv.Velocity = direction.Unit * speed
                else
                    bv.Velocity = Vector3.zero
                end
            end
        else
            table.remove(FollowMobs, i)
        end
    end
end)

function ReleaseMobs()
    for _, mob in ipairs(FollowMobs) do
        local hrp = mob:FindFirstChild("HumanoidRootPart")
        if hrp then
            local bv = hrp:FindFirstChild("BringBV")
            if bv then bv:Destroy() end
        end
    end
    FollowMobs = {}
end

RunService.Heartbeat:Connect(function(dt)
    if LocalPlayer.Character then
        local playerHrp = LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        if playerHrp then
            for i = #FollowMobs, 1, -1 do
                local mob = FollowMobs[i]
                if mob and mob.Parent and mob:FindFirstChild("HumanoidRootPart") and mob:FindFirstChild("Humanoid") and mob.Humanoid.Health > 0 then
                    local bv = mob.HumanoidRootPart:FindFirstChild("BringBV")
                    if bv then
                        local direction = (playerHrp.Position - Vector3.new(0, distance, 0) - mob.HumanoidRootPart.Position)
                        local mobDistance = direction.Magnitude
                        
                        -- Slower and smoother velocity, max speed 100
                        local velocityMagnitude = math.min(mobDistance * 2, 300)
                        bv.Velocity = direction.Unit * velocityMagnitude
                    end
                else
                    -- Remove dead or invalid mobs
                    table.remove(FollowMobs, i)
                end
            end
        end
    end
end)

-- Enhanced Damage Aura with better reliability
function DamageAura()
    if not DamageAuraEnabled then return end
    
    local char = LocalPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return end
    
    local playerHrp = char.HumanoidRootPart
    local currentTime = tick()
    local randomDelay = math.random(1, 30) / 100

    if currentTime - lastAttackTime < randomDelay then
        return
    end

    local targetsInRange = {}

    task.spawn(function()
    for _, mob in pairs(workspace.Enemies:GetChildren()) do
            if mob:FindFirstChild("Humanoid") and mob.Humanoid.Health > 0 and mob:FindFirstChild("HumanoidRootPart") then
                local dist = (playerHrp.Position - mob.HumanoidRootPart.Position).Magnitude
                if dist <= DamageAuraDistance then
                    table.insert(targetsInRange, {target = mob, dist = dist})
                end
            end
        end
    end)
    task.spawn(function()
    for _, player in pairs(game:GetService("Players"):GetPlayers()) do
            if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                if player.Character.Humanoid.Health > 0 then
                    local dist = (playerHrp.Position - player.Character.HumanoidRootPart.Position).Magnitude
                    if dist <= DamageAuraDistance then
                        table.insert(targetsInRange, {target = player.Character, dist = dist})
                    end
                end
            end
        end
    end)
    if #targetsInRange > 0 then
        lastAttackTime = currentTime
        
        pcall(function()
            game:GetService("ReplicatedStorage").Modules.Net["RE/RegisterAttack"]:FireServer(0.5)
        end)
       
        table.sort(targetsInRange, function(a, b)
            return a.dist < b.dist
        end)

        for _, data in ipairs(targetsInRange) do
            local target = data.target
            task.spawn(function()
                local bodyParts = {"HumanoidRootPart", "Head"}
                for _, partName in ipairs(bodyParts) do
                    local part = target:FindFirstChild(partName)
                    if part then
                        game:GetService("ReplicatedStorage").Modules.Net["RE/RegisterHit"]:FireServer(
                            part,
                            {playerHrp.Position},
                            tostring(math.random(100000, 999999)) .. "-" .. tostring(tick())
                        )
                    end
                end
            end)
        end
    end
end

-- Damage Aura Loop with faster check rate
task.spawn(function()

        if DamageAuraEnabled then
            local success, err = pcall(DamageAura)
            if not success then

        end
    end
end)

function Auto_Farm(tool)
    local char = LocalPlayer.Character
    if not char then return end

    local held_tool = char:FindFirstChildOfClass("Tool")
    if not held_tool or held_tool.ToolTip ~= tool then
        equip_weapon(tool)
    end

    if auto_buso then
        if char and not char:FindFirstChild("HasBuso") then
            Remotes.CommF_:InvokeServer("Buso")
        end
    end

    local currentData = getCurrentLevelData()
    if not currentData then 
        warn("No level data found for level: " .. level)
        return
    end

    -- Clean up dead mobs from FollowMobs table
    for i = #FollowMobs, 1, -1 do
        local mob = FollowMobs[i]
        if not mob or not mob.Parent or not mob:FindFirstChild("Humanoid") or mob.Humanoid.Health <= 0 then
            table.remove(FollowMobs, i)
        end
    end

    local target = nil
    local allMobs = {}

    -- Collect all alive mobs
    for _, v in pairs(workspace.Enemies:GetChildren()) do
        if v.Name == currentData.mob and v:FindFirstChild("Humanoid") and v.Humanoid.Health > 0 and v:FindFirstChild("HumanoidRootPart") then
            table.insert(allMobs, v)
            
            if bring_mob then
                if not table.find(FollowMobs, v) then
                    BringMob(v)
                end
            end
        end
    end

    -- Select target (prefer alive mobs)
    if #allMobs > 0 then
        target = allMobs[1]
    end

    if target then
        setTarget(target.HumanoidRootPart.Position + Vector3.new(0, distance, 0))
        if bring_mob then
            FollowMob = target
        end
        if not isTweenEnabled then
            startTweenLoop()
            isTweenEnabled = true
            enabled_highlight()
        end
    else
        setTarget(currentData.waitPos)
        FollowMob = nil
        if not isTweenEnabled then
            startTweenLoop()
            isTweenEnabled = true
            enabled_highlight()
        end
    end
end

function AutoQuest()
    local currentData = getCurrentLevelData()
    if not currentData then return end
    
    return game:GetService("ReplicatedStorage").Remotes.CommF_:InvokeServer(
        "StartQuest",
        currentData.quest,
        currentData.questNum
    )
end
